        private readonly SensorBias _rightBias = new();

        public SkiDataProcessor()

            _turnDetector.OnTurnCompleted += OnTurnCompleted;

        private void CalculateCalibration(SkiDataPoint point) {
            if (!_isCalibrated)
            {
                //ref SensorBias bias = ref ((point.Side == SensorSide.Left) ? ref _leftBias : ref _rightBias);
                SensorBias bias = point.Side == SensorSide.Left ? _leftBias : _rightBias;
                List<double> samples = point.Side == SensorSide.Left ? _leftSamples : _rightSamples;

                bias.AccX += point.AccX;
                bias.AccY += point.AccY;
                bias.AccZ += point.AccZ;
                bias.GyroZ += point.GyroZ;

                samples.Add(point.GyroZ);
                //System.Diagnostics.Trace.WriteLine($"Adding Calibrating point: {left.ToString()} ");
                //System.Diagnostics.Trace.WriteLine($"Adding Calibrating point: {right.ToString()} ");
                if (point.Side == SensorSide.Left) _leftSampleCount++; else _rightSampleCount++;

                if (_leftSampleCount >= SamplesRequired && _rightSampleCount >= SamplesRequired)
                {
                    // 1. Medie Accelerometro (Offset reale)
                    _leftBias.AccX /= SamplesRequired;
                    _leftBias.AccY /= SamplesRequired;
                    _leftBias.AccZ /= SamplesRequired;
                    // 2. MEDIA GIROSCOPIO (Fondamentale per lo zero del Roll)
                    _leftBias.GyroZ /= SamplesRequired;

                    _rightBias.AccX /= SamplesRequired;
                    _rightBias.AccY /= SamplesRequired;
                    _rightBias.AccZ /= SamplesRequired;
                    _rightBias.GyroZ /= SamplesRequired;

                    // 3. Calcolo dei Gain (opzionale se usi solo la sottrazione nel Calibrate)
                    double magL = Math.Sqrt(_leftBias.AccX * _leftBias.AccX + _leftBias.AccY * _leftBias.AccY + _leftBias.AccZ * _leftBias.AccZ);
                    double magR = Math.Sqrt(_rightBias.AccX * _rightBias.AccX + _rightBias.AccY * _rightBias.AccY + _rightBias.AccZ * _rightBias.AccZ);

                    _leftGain = 1.0 / magL;
                    _rightGain = 1.0 / magR;

                    _isCalibrated = true;
                }
            }
        }

        public void Process(SkiDataPoint point)
        {
            /*if (!_isCalibrated)
            {
                CalculateCalibration(point);
                return;
            }
            point = Calibrate(point);*/
            point = Smooth(point);

            if (point.Side == SensorSide.Left)
            {
                _leftDataPoint = point; 
            }
            else if (point.Side == SensorSide.Right)
            {
                _rightDataPoint = point;
            }

            // 2. Controllo: se manca uno dei due, esci con null (SENZA RESETTARE)
            if (_leftDataPoint == null || _rightDataPoint == null) return;

            // Aggiungiamo la metrica attualmente selezionata nella UI
            //System.Diagnostics.Trace.WriteLine($"Adding point: {point.ToString()} ");
            _turnDetector.AddPoints(_leftDataPoint, _rightDataPoint);

            _leftDataPoint = null;
            _rightDataPoint = null;
        }

        public SkiDataPoint Calibrate(SkiDataPoint point)
        {
            SensorBias bias = point.Side == SensorSide.Left ? _leftBias : _rightBias;

            // Sottraiamo l'offset medio misurato a riposo
            point.AccX = (float)(point.AccX - bias.AccX);
            point.AccY = (float)(point.AccY - bias.AccY);
            // Z: togliamo il bias e ridiamo 1.0 (gravità)
            point.AccZ = (float)(point.AccZ - bias.AccZ + 1.0);

            // Giroscopio: usiamo Z per il Roll
            point.GyroZ = (float)(point.GyroZ - bias.GyroZ);

            return point;
        }

        private void UpdateBuffer(List<double> b, double newValue, double sensitivityMultiplier = 1.5, double minThreshold = 0.2)
        {
            if (b.Count > 0)
            {
                // Ordiniamo e prendiamo la mediana attuale
                double currentMedian = b.OrderBy(x => x).ElementAt(b.Count / 2);

                // La soglia è: il doppio della mediana (o multiplier), 
                // ma MAI inferiore a minThreshold.
                double maxAllowedDiff = Math.Max(minThreshold, Math.Abs(currentMedian) * (sensitivityMultiplier - 1));

                if (Math.Abs(newValue - currentMedian) > maxAllowedDiff)
                {
                    // Se è un'allucinazione (il famoso 10G), inseriamo la mediana nel buffer
                    b.Add(currentMedian);
                }
                else
                {
                    b.Add(newValue);
                }
            }
            else
            {
                b.Add(newValue);
            }

            if (b.Count > PreSmoothWindow) b.RemoveAt(0);
        }

        private SkiDataPoint Smooth(SkiDataPoint point)

            {

                Roll = point.Roll;
                IsInitialized = true;

            point.SmoothAccY = AccY + Alpha * (point.AccY - AccY);
            point.SmoothAccZ = AccZ + Alpha * (point.AccZ - AccZ);
            point.SmoothRoll = Roll + Alpha * (point.Roll - Roll);
            //System.Diagnostics.Debug.WriteLine($"[DETECTOR] Point: {point.Timestamp} - {point.Side} - RawGyroz:{point.GyroZ}");

            return point;

        public void setMetrics(HashSet<MonitorMode> metrics)
        {

            _turnDetector.RequestedMetrics.Clear();
            foreach (MonitorMode metric in metrics )
            {

                _turnDetector.RequestedMetrics.Add(metric);
            }
        }

        private void OnTurnCompleted(object? sender, SkiTurn turn)
        {
            OnNewTurn?.Invoke(this, turn);
        }
